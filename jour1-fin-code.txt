import express from "express";

import { createHivesRouter } from "./routes/hives/hives.router.js";
import { createHivesController } from "./controllers/hives.controller.js";

import { createHivesService } from "./services/hives.service.js";
import { createMeasuresService } from "./services/measures.service.js";
import { createInspectionsService } from "./services/inspections.service.js";

import { requestLogger } from "./middleware/requestLogger.js";
import {
  notFoundHandler,
  errorHandler,
} from "./middlewares/errorHandler.js";

export function createApp({
  hivesRepository,
  measuresRepository,
  inspectionsRepository,
}) {
  const app = express();

  // Middlewares
  app.use(express.json());
  app.use(requestLogger);

  // Services
  const hivesService = createHivesService({
    hivesRepository,
  });

  const measuresService = createMeasuresService({
    hivesService,
    measuresRepository,
  });

  const inspectionsService = createInspectionsService({
    hivesService,
    inspectionsRepository,
  });

  // Controller
  const hivesController = createHivesController({
    hivesService,
    measuresService,
    inspectionsService,
  });

  // Routes
  app.get("/health", (req, res) => {
    res.status(200).json({ status: "ok" });
  });

  app.use("/api/hives", createHivesRouter(hivesController));

  // Gestion des erreurs : toujours à la fin
  app.use(notFoundHandler);
  app.use(errorHandler);

  return app;
}

import { sendProblem } from "../utils/problem.js";

// Route inconnue : un 404 au format RFC 9457, au lieu de la page HTML d'Express.
export function notFoundHandler(req, res) {
  sendProblem(res, 404, `Aucune route ne correspond à ${req.method} ${req.originalUrl}.`);
}

// Filet de sécurité : toute erreur non traitée arrive ici. Les quatre paramètres sont
// obligatoires : c'est à eux qu'Express reconnaît un intergiciel d'erreur.
export function errorHandler(err, req, res, next) {
  if (err.type === "entity.parse.failed") {
    return sendProblem(res, 400, "Le corps de la requête n'est pas un JSON valide.");
  }
  console.error(err); // la pile va dans le journal du serveur, jamais dans la réponse
  sendProblem(res, 500, "Une erreur interne est survenue.");
}

import { NotFoundError } from "../errors.js";

export function createHivesService({ hivesRepository }) {
  function getAllHives(filters = {}) {
    return hivesRepository.findAll(filters);
  }

  function getHiveById(id) {
    const hive = hivesRepository.findById(id);

    if (!hive) {
      throw new NotFoundError(
        `La ruche ${id} n'existe pas.`
      );
    }

    return hive;
  }

  return {
    getAllHives,
    getHiveById,
  };
}

import { ValidationError } from "../errors.js";

const validStatuses = ["forte", "normale", "faible", "morte"];

export function createInspectionsService({
  hivesService,
  inspectionsRepository,
}) {
  function addInspection(hiveId, data) {
    hivesService.getHiveById(hiveId);

    if (
      typeof data.id !== "string" ||
      !/^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(data.id)
    ) {
      throw new ValidationError("id doit être un UUID valide.");
    }

    if (
      !data.inspected_at ||
      Number.isNaN(Date.parse(data.inspected_at))
    ) {
      throw new ValidationError(
        "inspected_at doit être une date ISO 8601 valide."
      );
    }

    if (!validStatuses.includes(data.colony_status)) {
      throw new ValidationError(
        "colony_status doit être forte, normale, faible ou morte."
      );
    }

    if (
      data.queen_present !== undefined &&
      typeof data.queen_present !== "boolean"
    ) {
      throw new ValidationError(
        "queen_present doit être un booléen."
      );
    }

    if (
      data.observations !== undefined &&
      typeof data.observations !== "string"
    ) {
      throw new ValidationError(
        "observations doit être un texte."
      );
    }

    return inspectionsRepository.create({
      ...data,
      hive_id: hiveId,
    });
  }

  function getInspections(hiveId) {
    hivesService.getHiveById(hiveId);

    return inspectionsRepository.findByHiveId(hiveId);
  }

  return {
    addInspection,
    getInspections,
  };
}

export function createHivesController({
  hivesService,
  measuresService,
  inspectionsService,
}) {
  return {
    getAllHives(req, res) {
      const hives = hivesService.getAllHives({
        site_id: req.query.site_id || undefined,
      });

      res.status(200).json(hives);
    },

    getHiveById(req, res) {
      const hive = hivesService.getHiveById(req.params.id);

      res.status(200).json(hive);
    },

    addMeasure(req, res) {
      const measure = measuresService.addMeasure(
        req.params.id,
        req.body
      );

      res.status(201).json(measure);
    },

    getMeasures(req, res) {
      const result = measuresService.getMeasures(
        req.params.id,
        req.query
      );

      res.status(200).json(result);
    },

    getInspections(req, res) {
      const inspections = inspectionsService.getInspections(
        req.params.id
      );

      res.status(200).json(inspections);
    },

    addInspection(req, res) {
      const inspection = inspectionsService.addInspection(
        req.params.id,
        req.body
      );

      res.status(201).json(inspection);
    },
  };
}

export function createHivesRepository(initialHives = []) {
  const hives = [...initialHives];

  return {
    findAll({ site_id } = {}) {
      if (site_id) {
        return hives.filter(hive => hive.site_id === site_id);
      }

      return [...hives];
    },

    findById(id) {
      return hives.find(hive => hive.id === id) ?? null;
    },

    exists(id) {
      return hives.some(hive => hive.id === id);
    },
  };
}

}

export function createMeasuresRepository(initialMeasures = []) {
  const measures = [...initialMeasures];

  return {
    findByHiveId(hiveId) {
      return measures
        .filter(measure => measure.hive_id === hiveId)
        .sort(
          (a, b) =>
            new Date(b.timestamp) - new Date(a.timestamp)
        );
    },

    create(measure) {
      measures.push(measure);
      return measure;
    },

    count() {
      return measures.length;
    },
  };
}

import { test } from "node:test";
import assert from "node:assert/strict";

import { createMeasuresService } from "../services/measures.service.js";
import { ValidationError } from "../errors.js";

test("refuse un poids hors de sa plage physique", async () => {

  // 1. Préparer
  const savedMeasures = [];

  const hivesService = {
    getHiveById(id) {
      if (id !== "SN-CASA-001") {
        throw new Error("Ruche inconnue");
      }

      return {
        id: "SN-CASA-001",
        site_id: "SN-CASA",
      };
    },
  };

  const measuresRepository = {
    create(measure) {
      savedMeasures.push(measure);
      return measure;
    },

    findByHiveId() {
      return savedMeasures;
    },
  };

  const measuresService = createMeasuresService({
    hivesService,
    measuresRepository,
  });

  // 2. Agir
  const invalidMeasure = {
    timestamp: "2026-10-31T10:00:00Z",
    weight_kg: 500,
  };

  // 3. Vérifier
  await assert.rejects(
    async () => {
      measuresService.addMeasure(
        "SN-CASA-001",
        invalidMeasure
      );
    },
    ValidationError
  );

  assert.equal(savedMeasures.length, 0);
});
